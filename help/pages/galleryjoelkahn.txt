Round Spiral by Joel Kahn  (June 2010)
++++See Code Here|
<code>
# Contributed by Joel Kahn
fastgraphics
cls
clg
r=0:g=0:b=0
x=250.0:y=250.0
m=1.0
graphsize 500, 500
while x>0.0 and x<500.0 and y>0.0 and y<500.0
   m=m+4.0/m
   x=x+cos(m)
   y=y+sin(m)
   r=r+1
   if r>255 then
      r=1
      g=g+1
   end if
   if g>255 then
      g=1
      b=b+1
   end if
   color r, g, b
   plot x,y
end while
refresh
</code>

{{:en:kahn_joel_square_spiral.png?200|}}
Square Spiral by Joel Kahn  (June 2010)
++++See Code Here|
<code>
# Contributed by Joel Kahn
fastgraphics
cls
clg
graphsize 512, 512
r=1:g=1:b=1
x=250
y=250
s=1
plot x, y
for a=1 to 511
   x1=x+s*a
   y1=y+s*a
   q=s*1
   g=((abs(g+a))%255)+1
   for xx=x to x1 step q
      b=((abs(b+xx))%255)+1
      color r, g, b
      plot xx, y
   next xx
   for yy=y to y1 step q
      r=((abs(r+yy))%255)+1
      color r, g, b
      plot x1, yy
   next yy
   x=x1
   y=y1
   s=-s
   r=1:g=1:b=1
   refresh
next a
</code>

Jagged Depths by Joel Kahn  (July 2010)
++++See Code Here|
<code>
cls
clg
w=600.0
h=600.0
graphsize w,h
x=w/2.0
y=h/2.0
dim a(6)
a={0.0,0.0,0.0,x,y,0.0}
s1=0.9101
ss1=0.000347
s2=0.9103
ss2=0.000353
pie=3.1416
q1=pie
qq1=1.631*pie
q2=q1
qq2=qq1+s2
cc=137.0
c1=cc
c2=-cc
while s1<>s2
   s1=s1-ss1
   q1=q1+qq1
   c1=c1+cc
   color c1
   stamp x,y,s1,q1,a
   s2=s2-ss2
   q2=q2-qq2
   c2=c2-cc
   color c2
   stamp x,y,s2,q2,a
end while
</code>

Stellar Cores by Joel Kahn  (July 2010)
++++See Code Here|
<code>
cls
clg
w=600.0
h=600.0
graphsize w,h
x=w/2.0
y=h/2.0

ss=0.999
q=0.0
qq=0.4
for d=0.0 to 255.0 step 0.06
   s=s*ss
   q=q+qq
   b=d
   r=s*290.0
   g=q%256.0
   color r,g,b
   stamp x,y,s,q,a
next d
</code>
